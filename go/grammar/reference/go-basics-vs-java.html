六条铁律

  1. 没有隐式类型转换：float64(i) 必须写全。
  2. 声明未使用的变量 / import 未使用的包 = 编译错误（Java 只是 warning）。
  3. 大写导出，小写包内私有：没有 public/private 关键字。
  4. 只有 for 一种循环；if/for/switch 不写括号。
  5. { 必须跟语句同一行（编译器按行插入分号）。
  6. 错误是返回值（error），不是异常；panic 只留给真正异常的情况。

Hello, World

Java
public class Main {
    public static void main(String[] args) {
        System.out.println("Hello, 世界");
    }
}
Go
package main

import "fmt"

func main() {
	fmt.Println("Hello, 世界")
}

变量与常量

var x int = 10        // 完整形式（类型在前、在后，对应 Java 的 int x = 10）
var y = 10            // 类型推断为 int
var a, b = 1, "hi"    // 一行多个
z := 10               // 短声明：只能在函数内用，最惯用

const Pi = 3.14159    // 常量，编译期确定，写法同 Java final

显式类型转换

i := 42
f := float64(i)       // 必须显式；var f float64 = i 是编译错误
u := uint(7)
s := string(65)       // 按码点转：s == "A"（与 Java 强转 char 不同）

函数与多返回值

Java
static int[] divmod(int a, int b) {
    return new int[]{a / b, a % b};
}
// 调用：int[] r = divmod(17, 5);
//      r[0], r[1]
Go
func divmod(a, b int) (int, int) {
	return a / b, a % b
}

q, r := divmod(17, 5) // q=3, r=2
_, r2 := divmod(17, 5) // _ 丢弃不要的值
func add(a, b int) int { return a + b }  // 参数类型简写：a, b int

func divmod2(a, b int) (q, r int) {      // 命名返回值
	q = a / b
	r = a % b
	return                          // 裸返回（小函数可用，长函数慎用）
}

if / for / switch

if score >= 90 {
	fmt.Println("优")
} else if score >= 60 {
	fmt.Println("及格")
} else {
	fmt.Println("不及格")
}

if n := x * 2; n > 100 {   // 带初始化语句：n 只在 if/else 内可见
	fmt.Println(n)
}
for i := 0; i < 5; i++ { }  // 三段式，同 Java for
for i < 5 { }               // 条件式，同 Java while
for { break }               // 无限循环，break 跳出
switch day {
case "周六", "周日":          // 多值 case，逗号分隔
	fmt.Println("休息")      // 默认自动 break，无需写
default:
	fmt.Println("上班")
}
// 需要穿透到下一 case 时，显式写 fallthrough
