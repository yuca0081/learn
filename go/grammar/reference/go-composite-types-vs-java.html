七条铁律

  1. append 必须接返回值：s = append(s, x)。
  2. nil map：读安全（零值），写 panic → 先 make。
  3. nil 切片可以直接 append（与 nil map 相反，对比记忆）。
  4. map 遍历顺序随机；需要顺序自己维护 key 切片。
  5. struct 赋值 / 传参 = 整体复制；想共享显式用 &。
  6. 方法要改字段 → 指针接收者；惯例：一个 struct 的接收者类型统一。
  7. 没有构造函数 → 惯用 NewXxx() *Xxx 函数。

切片

s := []int{90, 85, 77}        // 字面量
s = append(s, 60)             // 追加（可多个：append(s, 1, 2, 3)）
s = append(s, other...)       // 拼接另一个切片（... 展开）
var s2 []int                  // nil 切片，可直接 append
nums := make([]int, 0, 100)   // 预分配容量 100
sub := s[1:3]                 // 左闭右开，共享底层数组！
fmt.Println(len(s))           // 长度

map

m := map[string]int{"a": 1}   // 字面量
m["b"] = 2                    // 写
n := m["zz"]                  // 缺失键 → 零值 0
v, ok := m["a"]               // comma-ok：ok=键是否存在
delete(m, "b")                // 删除
m2 := make(map[string]int)    // 空 map（struct 字段必用这个）

struct / 指针 / 方法

type Player struct {
	name string       // 小写：包内私有
	hp   int
}

p1 := Player{name: "勇者", hp: 100}   // 字面量（字段名写全）
p2 := p1                              // 复制！改 p2 不影响 p1
p3 := &p1                             // *Player 指针，共享
p3.hp = 80                            // 自动解引用，改的就是 p1

func (p Player) Name() string { return p.name }  // 值接收者：只读副本
func (p *Player) Heal(n int) { p.hp += n }       // 指针接收者：能改原值
// 调用一样：p.Heal(20)，Go 自动取地址

func NewPlayer(name string) *Player {             // 构造器惯例
	return &Player{name: name, hp: 100}
}

写法用途
for i, v := range s切片：下标 + 值
for _, v := range s只要值
for i := range s只要下标
for k, v := range mmap：键 + 值（顺序随机）
for k := range m只要键
